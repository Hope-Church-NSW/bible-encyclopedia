async function boot(){ try{ setStatus("جاري تحميل قاعدة البيانات (حجم الملف 7.5MB)..."); const r=await fetch(DATA_URL,{cache:"reload"}); if(!r.ok) throw new Error("HTTP "+r.status); const data=await r.json(); if(!Array.isArray(data)) throw new Error("قاعدة البيانات ليست مصفوفة JSON"); verses=data.filter(v=>v && v.book && v.chapter && v.verse && v.text); books=uniqueBooks(verses); fillBooks(); $("count").textContent=`${verses.length.toLocaleString("ar-EG")} سجل`; setStatus("تم تحميل قاعدة الكتاب المقدس بنجاح."); if(books.length){ $("book").value=books[0]; fillChapters(); $("chapter").value="1"; renderChapter(); } }catch(e){ setStatus("تعذر تحميل bible_database.json. يرجى إعادة تحديث الصفحة (Ctrl + F5)."); $("count").textContent="خطأ في التحميل"; console.error(e); } }
