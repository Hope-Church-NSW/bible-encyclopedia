// ============================================================ // lib/features/bible/services/verse_context_service.dart // ============================================================ import '../../../core/database/app_database.dart'; class VerseContext { final int verseId; final List crossReferences; final List studyIds; final List sourceIds; final List wordIds; const VerseContext({ required this.verseId, this.crossReferences = const [], this.studyIds = const [], this.sourceIds = const [], this.wordIds = const [], }); } class VerseContextService { final AppDatabase database; VerseContextService( this.database, ); Future getContext( int verseId, ) async { final db = await database.database; final crossReferences = await db.query( 'cross_references', where: 'source_verse_id = ?', whereArgs: [ verseId, ], ); final words = await db.query( 'words', where: 'verse_id = ?', whereArgs: [ verseId, ], ); return VerseContext( verseId: verseId, crossReferences: crossReferences .map( (row) => row['target_verse_id'] as int, ) .toList(), wordIds: words .map( (row) => row['id'] as int, ) .toList(), ); } }
